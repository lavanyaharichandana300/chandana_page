<script>
// --- Navigation ---
function go(id){
  document.querySelectorAll('.section').forEach(s=>s.style.display="none");
  document.getElementById(id).style.display="block";
}

// --- Buttons ---
const yesBtn=document.getElementById("yesBtn");
const noBtn=document.getElementById("noBtn");
const restartBtn1=document.getElementById("restartBtn1");
const restartBtn2=document.getElementById("restartBtn2");
const themeBtn=document.getElementById("themeBtn");

yesBtn.addEventListener("click",()=>{ go('p2'); createHearts(); continuousSparkles(); });
restartBtn1.addEventListener("click",()=>{ go('p1'); });
restartBtn2.addEventListener("click",()=>{ go('p1'); });
themeBtn.addEventListener("click",()=>{ document.body.classList.toggle("dark"); });

// --- Hearts & Sparkles ---
function createHearts(){
  for(let i=0;i<12;i++){
    const h=document.createElement("div"); h.className="heart"; h.innerText="💖";
    h.style.left=Math.random()*window.innerWidth+"px";
    document.body.appendChild(h);
    setTimeout(()=>h.remove(),2000);
  }
}

function continuousSparkles(){
  for(let i=0;i<20;i++){
    const s=document.createElement("div"); s.className="sparkle";
    s.innerText=["💖","✨","🌸","🌟"][Math.floor(Math.random()*4)];
    s.style.left=Math.random()*window.innerWidth+"px";
    s.style.fontSize=(10+Math.random()*30)+"px";
    document.body.appendChild(s);
    setTimeout(()=>s.remove(),4000);
  }
}

// --- Secret Easter Egg ---
let tapCount=0; let tapTimeout;
document.getElementById("cat1").addEventListener("click",()=>{
  tapCount++;
  clearTimeout(tapTimeout);
  if(tapCount>=5){
    for(let i=0;i<40;i++){
      const s=document.createElement("div"); s.className="sparkle";
      s.innerText=["🎉","✨","💥","🌟","💖"][Math.floor(Math.random()*5)];
      s.style.left=Math.random()*window.innerWidth+"px";
      s.style.fontSize=(15+Math.random()*35)+"px";
      document.body.appendChild(s);
      setTimeout(()=>s.remove(),3000);
    }
    alert("🎊 Secret Cat Party! Rahul, Chandana loves you! 🎊");
    tapCount=0; return;
  }
  tapTimeout=setTimeout(()=>tapCount=0,1500);
});

// --- NO button runaway ---
let noClicks=0;
noBtn.addEventListener("mouseenter",()=>{
  noClicks++;
  if(noClicks<=5){
    const x=Math.random()*60-30;
    const y=Math.random()*40-20;
    noBtn.style.transform=`translate(${x}px,${y}px)`;
  } else {
    noBtn.style.transform=`translate(0,0)`;
    go('p3');
  }
});

// --- Start ---
go('p1');
